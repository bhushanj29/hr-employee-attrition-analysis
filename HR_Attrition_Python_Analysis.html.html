# Attrition by Department
attrition_dept = df.groupby("Department")["Attrition"].apply(
    lambda x: (x=="Yes").sum()).sort_values(ascending = False) 

attrition_dept.plot(kind="bar",figsize =(10,5))
plt.title("Employees Attrition By Department")
plt.xlabel("Department")
plt.ylabel("Employees Left")
plt.xticks(rotation = 45)
plt.show()
